async function analyzeReceiptWithGemini(base64Data, apiKey) { const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent`; const prompt = `אתה עוזר מומחה לסריקת קבלות מסופרמרקטים בישראל. עיין בתמונה הזו של הקבלה וחלץ מתוכה את המידע הבא במבנה JSON בלבד (ללא קוד markdown וללא הסברים נוספים): { "store_name": "שם הסופר/הרשת (למשל: אושר עד, שופרסל, רמי לוי)", "purchase_date": "YYYY-MM-DD", "total_amount": 0.00, "items": [ { "name": "שם המוצר", "price": 0.00 } ] }`; const payload = { contents: [ { parts: [ { text: prompt }, { inline_data: { mime_type: "image/jpeg", data: base64Data } } ] } ] }; const headers = { 'Content-Type': 'application/json' }; // אם המפתח מתחיל ב-AQ, נשלח אותו כ-Bearer Token יחד עם מזהה הפרויקט שלך if (apiKey.startsWith('AQ.')) { headers['Authorization'] = `Bearer ${apiKey}`; headers['x-goog-user-project'] = '702330486389'; } const fetchUrl = apiKey.startsWith('AQ.') ? url : `${url}?key=${apiKey}`; const response = await fetch(fetchUrl, { method: 'POST', headers: headers, body: JSON.stringify(payload) }); if (!response.ok) { const errText = await response.text(); console.error('API Error details:', errText); throw new Error(`תקלת API (${response.status}): ${errText}`); } const result = await response.json(); if (!result.candidates || result.candidates.length === 0) { throw new Error('לא התקבלה תשובה תקינה מ-Gemini'); } let textResponse = result.candidates[0].content.parts[0].text; textResponse = textResponse.replace(/```json/g, '').replace(/```/g, '').trim(); return JSON.parse(textResponse); }
