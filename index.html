async function analyzeReceiptWithGemini(base64Data, apiKey) { // רשימת מודלים לניסיון במקרה של 404 (מהחדש והמומלץ לישן) const models = ['gemini-2.5-flash', 'gemini-2.0-flash', 'gemini-1.5-flash']; const prompt = `אתה עוזר מומחה לסריקת קבלות מסופרמרקטים בישראל. עיין בתמונה הזו של הקבלה וחלץ מתוכה את המידע הבא במבנה JSON בלבד (ללא קוד markdown וללא הסברים נוספים): { "store_name": "שם הסופר/הרשת (למשל: אושר עד, שופרסל, רמי לוי)", "purchase_date": "YYYY-MM-DD", "total_amount": 0.00, "items": [ { "name": "שם המוצר", "price": 0.00 } ] }`; const payload = { contents: [ { parts: [ { text: prompt }, { inline_data: { mime_type: "image/jpeg", data: base64Data } } ] } ] }; let lastError = null; for (const model of models) { try { const isAQKey = apiKey.startsWith('AQ.'); const baseUrl = `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent`; const headers = { 'Content-Type': 'application/json' }; let fetchUrl = baseUrl; if (isAQKey) { headers['Authorization'] = `Bearer ${apiKey}`; headers['x-goog-user-project'] = '702330486389'; } else { fetchUrl = `${baseUrl}?key=${apiKey}`; } console.log(`מנסה לפנות למודל ${model}...`); const response = await fetch(fetchUrl, { method: 'POST', headers: headers, body: JSON.stringify(payload) }); if (response.ok) { const result = await response.json(); if (result.candidates && result.candidates[0]?.content?.parts[0]?.text) { let textResponse = result.candidates[0].content.parts[0].text; textResponse = textResponse.replace(/```json/g, '').replace(/```/g, '').trim(); return JSON.parse(textResponse); } } else { const errText = await response.text(); console.warn(`כישלון במודל ${model} (סטטוס ${response.status}):`, errText); lastError = `שגיאה ${response.status}: ${errText}`; } } catch (e) { console.error(`שגיאה בחיבור למודל ${model}:`, e); lastError = e.message; } } throw new Error(`כל ניסיונות החיבור ל-Gemini נכשלו. שגיאה אחרונה: ${lastError}`); }
